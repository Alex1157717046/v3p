#   E2: 3 = NC(X); 13 = GND(绿线后垂直线向上接 GND 符号); 12..1 =
#       MOTOR_EN(+红色全局标签 PA8), M1_IN3, M1_IN2, M1_IN1, M0_IN3, M0_IN2, M0_IN1,
